  <script>
    // State management
    let currentStep = 1;
    let unlockedSteps = [true, false, false];
    let currentEye = 'left'; // 'left' or 'right'
    let earthPos = 'A'; // 'A' or 'B'

    // Canvas contexts
    const c1 = document.getElementById('canvas1');
    const ctx1 = c1.getContext('2d');
    const c2 = document.getElementById('canvas2');
    const ctx2 = c2.getContext('2d');
    const c3 = document.getElementById('canvas3');
    const ctx3 = c3.getContext('2d');

    // Resize canvas for sharp rendering
    function initCanvasSize() {
      [c1, c2, c3].forEach(canvas => {
        canvas.width = canvas.parentElement.clientWidth;
        canvas.height = canvas.parentElement.clientHeight;
      });
      drawStep1();
      drawStep2();
      drawStep3();
    }
    window.addEventListener('resize', initCanvasSize);

    // --- STEP 1: Observer Perspective ---
    function setEye(eye) {
      currentEye = eye;
      drawStep1();
    }

    function drawStep1() {
      const w = c1.width;
      const h = c1.height;
      ctx1.clearRect(0, 0, w, h);

      // Background stars & mountains
      ctx1.fillStyle = "#ffffff";
      for(let i=0; i<30; i++) {
        let x = (i * 37) % w;
        let y = (i * 19) % (h * 0.5);
        ctx1.fillRect(x, y, 2, 2);
      }

      // Background mountains
      ctx1.fillStyle = "#1e293b";
      ctx1.beginPath();
      ctx1.moveTo(0, h);
      ctx1.lineTo(w * 0.2, h * 0.5);
      ctx1.lineTo(w * 0.5, h);
      ctx1.lineTo(w * 0.8, h * 0.4);
      ctx1.lineTo(w, h);
      ctx1.fill();

      // Near Target Star (Position shifts based on eye view)
      // Left eye view -> Star appears on the Right side relative to background
      // Right eye view -> Star appears on the Left side
      let starX = (currentEye === 'left') ? w * 0.65 : w * 0.35;
      let starY = h * 0.5;

      // Line of Sight indicator
      ctx1.strokeStyle = "rgba(234, 179, 8, 0.4)";
      ctx1.setLineDash([5, 5]);
      ctx1.beginPath();
      ctx1.moveTo(w/2, h);
      ctx1.lineTo(starX, starY);
      ctx1.stroke();
      ctx1.setLineDash([]);

      // Draw Near Yellow Star
      ctx1.fillStyle = "#facc15";
      ctx1.shadowColor = "#facc15";
      ctx1.shadowBlur = 15;
      ctx1.beginPath();
      ctx1.arc(starX, starY, 16, 0, Math.PI * 2);
      ctx1.fill();
      ctx1.shadowBlur = 0;

      // Label
      ctx1.fillStyle = "#ffffff";
      ctx1.font = "bold 14px sans-serif";
      ctx1.textAlign = "center";
      ctx1.fillText("가까운 별", starX, starY - 25);
      
      ctx1.fillStyle = "#38bdf8";
      ctx1.fillText(`현재 관측: [ ${currentEye === 'left' ? '왼쪽 눈' : '오른쪽 눈'} ]`, w/2, 30);
    }

    // --- STEP 2: Top-down View ---
    function toggleEarthPos() {
      earthPos = (earthPos === 'A') ? 'B' : 'A';
      drawStep2();
    }

    function drawStep2() {
      const w = c2.width;
      const h = c2.height;
      ctx2.clearRect(0, 0, w, h);

      const starX = w / 2;
      const starY = h * 0.45;

      const posA = { x: w * 0.25, y: h * 0.85 };
      const posB = { x: w * 0.75, y: h * 0.85 };

      // Background Sky Wall (Top)
      ctx2.strokeStyle = "#475569";
      ctx2.lineWidth = 3;
      ctx2.beginPath();
      ctx2.moveTo(20, 40);
      ctx2.lineTo(w - 20, 40);
      ctx2.stroke();
      
      ctx2.fillStyle = "#94a3b8";
      ctx2.font = "12px sans-serif";
      ctx2.textAlign = "center";
      ctx2.fillText("배경 천구 (멀리 있는 별들)", w/2, 25);

      // Light Rays to Background Wall
      // Ray from A through Star to Wall
      let projAX = starX + (starX - posA.x) * (starY - 40) / (posA.y - starY);
      ctx2.strokeStyle = (earthPos === 'A') ? "#22c55e" : "#334155";
      ctx2.lineWidth = (earthPos === 'A') ? 2 : 1;
      ctx2.beginPath();
      ctx2.moveTo(posA.x, posA.y);
      ctx2.lineTo(projAX, 40);
      ctx2.stroke();

      // Ray from B through Star to Wall
      let projBX = starX - (posB.x - starX) * (starY - 40) / (posB.y - starY);
      ctx2.strokeStyle = (earthPos === 'B') ? "#22c55e" : "#334155";
      ctx2.lineWidth = (earthPos === 'B') ? 2 : 1;
      ctx2.beginPath();
      ctx2.moveTo(posB.x, posB.y);
      ctx2.lineTo(projBX, 40);
      ctx2.stroke();

      // Projections on wall
      ctx2.fillStyle = "#facc15";
      ctx2.beginPath();
      ctx2.arc(projAX, 40, 6, 0, Math.PI * 2);
      ctx2.arc(projBX, 40, 6, 0, Math.PI * 2);
      ctx2.fill();

      ctx2.fillStyle = "#ffffff";
      ctx2.font = "11px sans-serif";
      ctx2.fillText("A에서 본 위치", projAX, 58);
      ctx2.fillText("B에서 본 위치", projBX, 58);

      // Target Star
      ctx2.fillStyle = "#ef4444";
      ctx2.shadowColor = "#ef4444";
      ctx2.shadowBlur = 10;
      ctx2.beginPath();
      ctx2.arc(starX, starY, 12, 0, Math.PI * 2);
      ctx2.fill();
      ctx2.shadowBlur = 0;
      
      ctx2.fillStyle = "#ffffff";
      ctx2.font = "bold 13px sans-serif";
      ctx2.fillText("가까운 별", starX + 35, starY + 4);

      // Earth Positions
      drawEarth(ctx2, posA.x, posA.y, "관측지점 A", earthPos === 'A');
      drawEarth(ctx2, posB.x, posB.y, "관측지점 B", earthPos === 'B');
    }

    function drawEarth(ctx, x, y, label, active) {
      ctx.fillStyle = active ? "#38bdf8" : "#64748b";
      ctx.beginPath();
      ctx.arc(x, y, 14, 0, Math.PI * 2);
      ctx.fill();
      if(active) {
        ctx.strokeStyle = "#ffffff";
        ctx.lineWidth = 2;
        ctx.stroke();
      }
      ctx.fillStyle = "#ffffff";
      ctx.font = "12px sans-serif";
      ctx.textAlign = "center";
      ctx.fillText(label, x, y + 28);
    }

    // --- STEP 3: Distance Dependency ---
    function drawStep3() {
      const w = c3.width;
      const h = c3.height;
      ctx3.clearRect(0, 0, w, h);

      const distInput = parseInt(document.getElementById('starDist').value);
      const labelText = document.getElementById('distVal');
      
      if(distInput < 150) labelText.innerText = "매우 가깝음";
      else if(distInput < 210) labelText.innerText = "보통 거리";
      else labelText.innerText = "멀음";

      const posA = { x: w * 0.3, y: h * 0.85 };
      const posB = { x: w * 0.7, y: h * 0.85 };
      
      // Star Y Position based on Slider (Lower Y = Closer to Earth)
      const starX = w / 2;
      const starY = posA.y - distInput;

      // Lines of Sight
      ctx3.strokeStyle = "#6366f1";
      ctx3.lineWidth = 2;
      ctx3.beginPath();
      ctx3.moveTo(posA.x, posA.y);
      ctx3.lineTo(starX, starY);
      ctx3.lineTo(posB.x, posB.y);
      ctx3.stroke();

      // Draw Parallax Angle Arc
      const angleA = Math.atan2(posA.y - starY, posA.x - starX);
      const angleB = Math.atan2(posB.y - starY, posB.x - starX);
      
      ctx3.strokeStyle = "#f43f5e";
      ctx3.lineWidth = 3;
      ctx3.beginPath();
      ctx3.arc(starX, starY, 35, angleA, angleB, false);
      ctx3.stroke();

      // Angle Label
      ctx3.fillStyle = "#f43f5e";
      ctx3.font = "bold 15px sans-serif";
      ctx3.textAlign = "center";
      ctx3.fillText("시차(각도)", starX, starY + 55);

      // Star
      ctx3.fillStyle = "#facc15";
      ctx3.shadowColor = "#facc15";
      ctx3.shadowBlur = 12;
      ctx3.beginPath();
      ctx3.arc(starX, starY, 12, 0, Math.PI * 2);
      ctx3.fill();
      ctx3.shadowBlur = 0;

      // Earth Positions
      drawEarth(ctx3, posA.x, posA.y, "관측위치 A", true);
      drawEarth(ctx3, posB.x, posB.y, "관측위치 B", true);
    }

    // --- Tab Navigation ---
    function switchTab(step) {
      if(!unlockedSteps[step - 1]) return;

      currentStep = step;
      document.querySelectorAll('.tab-btn').forEach((btn, idx) => {
        btn.classList.toggle('active', idx === step - 1);
      });
      document.querySelectorAll('.content-box').forEach((box, idx) => {
        box.classList.toggle('active', idx === step - 1);
      });

      setTimeout(initCanvasSize, 50);
    }

    function unlockAndGo(nextStep) {
      unlockedSteps[nextStep - 1] = true;
      
      const tab = document.getElementById(`tab${nextStep}`);
      const badge = document.getElementById(`badge${nextStep}`);
      
      tab.classList.add('unlocked');
      badge.classList.add('pass');
      badge.innerText = "진행 가능";

      switchTab(nextStep);
    }

    // --- AI Concept Grading Logic (NLP Rule Engine) ---
    function checkAnswer(step) {
      const input = document.getElementById(`ans${step}`).value.trim().toLowerCase();
      const feedback = document.getElementById(`feedback${step}`);
      const nextBtn = document.getElementById(`next${step}`);

      if(!input) {
        feedback.className = "feedback-msg error";
        feedback.innerText = "⚠️ 답안을 입력해 주세요!";
        return;
      }

      let isCorrect = false;

      if(step === 1) {
        // Correct concepts: "더 많이", "많이", "크게", "더 크게"
        const keywords = ["많이", "크게", "더"];
        const wrongKeywords = ["적게", "작게", "안", "가만히"];
        
        const hasWrong = wrongKeywords.some(w => input.includes(w));
        const hasRight = keywords.some(k => input.includes(k));

        if(hasRight && !hasWrong) isCorrect = true;

      } else if(step === 2) {
        // Correct concepts: "관측자", "위치", "눈", "지구", "보는", "시선"
        const keywords = ["관측", "위치", "눈", "지구", "보는", "시선", "장소", "거리"];
        const wrongKeywords = ["별이", "밝기", "태양"];

        const hasRight = keywords.some(k => input.includes(k));
        if(hasRight) isCorrect = true;

      } else if(step === 3) {
        // Correct concepts: "커진다", "크다", "증가", "넓어진다"
        const keywords = ["커", "크", "증가", "넓"];
        const wrongKeywords = ["작", "줄어", "감소", "일정"];

        const hasWrong = wrongKeywords.some(w => input.includes(w));
        const hasRight = keywords.some(k => input.includes(k));

        if(hasRight && !hasWrong) isCorrect = true;
      }

      if(isCorrect) {
        feedback.className = "feedback-msg success";
        feedback.innerText = "✨ AI 채점 결과: 정답입니다! 개념을 올바르게 이해하셨습니다.";
        
        if(step < 3) {
          nextBtn.style.display = "block";
        } else {
          document.getElementById("finalSuccess").style.display = "block";
          document.getElementById("badge3").classList.add("pass");
          document.getElementById("badge3").innerText = "완료 🏆";
        }
      } else {
        feedback.className = "feedback-msg error";
        feedback.innerText = "❌ AI 채점 결과: 정답이 아닙니다. 다시 한번 고민해보세요!";
      }
    }
